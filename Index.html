      initContext() {

      // Subtle, tactile wooden block click/pop
      playPop() {
        try {
          this.initContext();
          const osc = this.ctx.createOscillator();
          const gain = this.ctx.createGain();

          osc.type = 'triangle';
          // Pitch drops fast to simulate resonant wood tap
          osc.frequency.setValueAtTime(540, this.ctx.currentTime);
          osc.frequency.exponentialRampToValueAtTime(140, this.ctx.currentTime + 0.05);

          gain.gain.setValueAtTime(0.2, this.ctx.currentTime);
          gain.gain.exponentialRampToValueAtTime(0.001, this.ctx.currentTime + 0.055);

          osc.connect(gain);
          gain.connect(this.ctx.destination);

          osc.start();
          osc.stop(this.ctx.currentTime + 0.06);
        } catch (e) {
          console.warn('Audio not allowed yet', e);
        }
      },

      // Gentle water splash/fizz on watering
      playWaterSplash() {
        try {
          this.initContext();
          const now = this.ctx.currentTime;
         
          // Noise burst through sweeping bandpass filter
          const bufferSize = this.ctx.sampleRate * 0.4;
          const noiseBuffer = this.ctx.createBuffer(1, bufferSize, this.ctx.sampleRate);
          const data = noiseBuffer.getChannelData(0);
          for (let i = 0; i < bufferSize; i++) {
            data[i] = Math.random() * 2 - 1;
          }

          const noise = this.ctx.createBufferSource();
          noise.buffer = noiseBuffer;

          const filter = this.ctx.createBiquadFilter();
          filter.type = 'bandpass';
          filter.frequency.setValueAtTime(600, now);
          filter.frequency.exponentialRampToValueAtTime(1800, now + 0.2);
          filter.frequency.exponentialRampToValueAtTime(300, now + 0.38);
          filter.Q.value = 3.0;

          const gain = this.ctx.createGain();
          gain.gain.setValueAtTime(0.25, now);
          gain.gain.exponentialRampToValueAtTime(0.001, now + 0.38);

          noise.connect(filter);
          filter.connect(gain);
          gain.connect(this.ctx.destination);

          // Harmonic bubbling tones
          [520, 680, 840].forEach((freq, i) => {
            const osc = this.ctx.createOscillator();
            const bGain = this.ctx.createGain();
            osc.type = 'sine';
            osc.frequency.setValueAtTime(freq, now + i * 0.07);
            osc.frequency.exponentialRampToValueAtTime(freq * 1.4, now + i * 0.07 + 0.08);

            bGain.gain.setValueAtTime(0.08, now + i * 0.07);
            bGain.gain.exponentialRampToValueAtTime(0.001, now + i * 0.07 + 0.08);

            osc.connect(bGain);
            bGain.connect(this.ctx.destination);
            osc.start(now + i * 0.07);
            osc.stop(now + i * 0.07 + 0.09);
          });

          noise.start(now);
          noise.stop(now + 0.4);
        } catch (e) {
          console.warn('Splash audio error', e);
        }
      },

      // Calming pentatonic wind chime harmony on Day Completion
      playChime() {
        try {
          this.initContext();
          const now = this.ctx.currentTime;
          // Meditative Pentatonic Frequencies (Eb Major pentatonic: Eb, F, G, Bb, C)
          const notes = [311.13, 392.00, 466.16, 523.25, 622.25];
         
          notes.forEach((freq, idx) => {
            const startT = now + idx * 0.08;
            const osc = this.ctx.createOscillator();
            const gain = this.ctx.createGain();

            osc.type = 'sine';
            osc.frequency.setValueAtTime(freq, startT);

            gain.gain.setValueAtTime(0.0001, startT);
            gain.gain.linearRampToValueAtTime(0.12, startT + 0.03);
            gain.gain.exponentialRampToValueAtTime(0.0001, startT + 1.8);

            osc.connect(gain);
            gain.connect(this.ctx.destination);

            osc.start(startT);
            osc.stop(startT + 1.9);
          });
        } catch (e) {
          console.warn('Chime audio error', e);
        }
      },

      // Continuous therapeutic rain / warm ambient noise loop
      toggleAmbient() {
        this.initContext();
        if (this.isAmbientPlaying) {
          this.stopAmbient();
        } else {
          this.startAmbient();
        }
        return this.isAmbientPlaying;
      },

      startAmbient() {
        try {
          const now = this.ctx.currentTime;
          this.ambientGain = this.ctx.createGain();
          this.ambientGain.gain.setValueAtTime(0.0001, now);
          this.ambientGain.linearRampToValueAtTime(0.12, now + 1.2);
          this.ambientGain.connect(this.ctx.destination);

          // 1. Soft Lowpass Pink/Brown Rain Sound
          const bufferSize = this.ctx.sampleRate * 3;
          const noiseBuffer = this.ctx.createBuffer(1, bufferSize, this.ctx.sampleRate);
          const data = noiseBuffer.getChannelData(0);
          let lastOut = 0.0;
          for (let i = 0; i < bufferSize; i++) {
            const white = Math.random() * 2 - 1;
            data[i] = (lastOut + (0.02 * white)) / 1.02;
            lastOut = data[i];
            data[i] *= 3.5;
          }

          this.noiseNode = this.ctx.createBufferSource();
          this.noiseNode.buffer = noiseBuffer;
          this.noiseNode.loop = true;

          const filter = this.ctx.createBiquadFilter();
          filter.type = 'lowpass';
          filter.frequency.value = 580;

          this.noiseNode.connect(filter);
          filter.connect(this.ambientGain);

          // 2. Meditative 432Hz harmonic drone oscillators (tuned low at 108Hz and 162Hz)
          this.droneOsc1 = this.ctx.createOscillator();
          this.droneOsc2 = this.ctx.createOscillator();
          this.droneOsc1.type = 'sine';
          this.droneOsc2.type = 'sine';
          this.droneOsc1.frequency.value = 108; // 432 / 4
          this.droneOsc2.frequency.value = 162; // 432 * 3/8

          const droneGain = this.ctx.createGain();
          droneGain.gain.value = 0.035;

          this.droneOsc1.connect(droneGain);
          this.droneOsc2.connect(droneGain);
          droneGain.connect(this.ambientGain);

          this.noiseNode.start(now);
          this.droneOsc1.start(now);
          this.droneOsc2.start(now);

          this.isAmbientPlaying = true;
        } catch (e) {
          console.warn('Ambient start failed', e);
        }
      },

      stopAmbient() {
        if (!this.isAmbientPlaying || !this.ambientGain) return;
        const now = this.ctx.currentTime;
        this.ambientGain.gain.linearRampToValueAtTime(0.0001, now + 0.8);
        setTimeout(() => {
          try {
            if (this.noiseNode) this.noiseNode.stop();
            if (this.droneOsc1) this.droneOsc1.stop();
            if (this.droneOsc2) this.droneOsc2.stop();
          } catch(e){}
          this.isAmbientPlaying = false;
        }, 850);
      }
    };

    function buildWordGrid(word) {
      const size = 36;
      const grid = new Array(size).fill(null);
      const chars = word.split('');
      const startPos = [0, 1, 6, 7, 12, 13][(state.day * 2) % 6];
     
      for (let i = 0; i < chars.length; i++) {
        grid[startPos + i] = chars[i];
      }

      const filler = currentLang === 'en' ? 'HARMONYSOULSEEDGROWTHLEAF' : 'TANAHBUNGAMEKARDAMAIJIWA';
      for (let i = 0; i < size; i++) {
        if (!grid[i]) {
          grid[i] = filler[(i * 7 + state.day) % filler.length];
        }
      }
      return grid;
    }

    function onLetterClick(index) {

      }

    // Architectural Pot Drawing Routine (Terracotta & Celestial)
    function drawPot(c, potStyle, x, y, width, height) {
      c.save();
      const topW = width;
      const botW = width * 0.72;
      const rimH = height * 0.24;

      if (potStyle === 'celestial') {
        // Celestial Soft Glow Aura
        c.shadowColor = 'rgba(165, 180, 252, 0.45)';
        c.shadowBlur = 18;

        // Pot Body Gradient (Deep Indigo)
        const pGrad = c.createLinearGradient(x - topW/2, y, x + topW/2, y + height);
        pGrad.addColorStop(0, '#161938');
        pGrad.addColorStop(0.5, '#232959');
        pGrad.addColorStop(1, '#0e1128');

        // Draw Trapezoid Body
        c.fillStyle = pGrad;
        c.beginPath();
        c.moveTo(x - topW * 0.46, y + rimH);
        c.lineTo(x + topW * 0.46, y + rimH);
        c.lineTo(x + botW * 0.5, y + height);
        c.lineTo(x - botW * 0.5, y + height);
        c.closePath();
        c.fill();

        // Celestial Gold Constellation Lines & Stars
        c.strokeStyle = 'rgba(253, 230, 138, 0.55)';
        c.lineWidth = 1;
        c.beginPath();
        c.moveTo(x - topW * 0.25, y + rimH + 10);
        c.lineTo(x - 8, y + height * 0.65);
        c.lineTo(x + 18, y + height * 0.55);
        c.lineTo(x + topW * 0.28, y + height * 0.75);
        c.stroke();

        // Star Dots
        c.fillStyle = '#fef08a';
        [
          { dx: -topW * 0.25, dy: rimH + 10 },
          { dx: -8, dy: height * 0.65 },
          { dx: 18, dy: height * 0.55 },
          { dx: topW * 0.28, dy: height * 0.75 },
          { dx: 0, dy: height * 0.82 }
        ].forEach(pt => {
          c.beginPath();
          c.arc(x + pt.dx, y + pt.dy, 2, 0, Math.PI * 2);
          c.fill();
        });

        // Gold Rim
        c.shadowBlur = 8;
        c.shadowColor = 'rgba(234, 179, 8, 0.4)';
        const rGrad = c.createLinearGradient(x - topW/2, y, x + topW/2, y);
        rGrad.addColorStop(0, '#ca8a04');
        rGrad.addColorStop(0.5, '#fef08a');
        rGrad.addColorStop(1, '#a16207');
        c.fillStyle = rGrad;
        c.beginPath();
        c.roundRect(x - topW * 0.52, y, topW * 1.04, rimH, [4, 4, 3, 3]);
        c.fill();

        // Drainage Saucer
        c.fillStyle = '#1e1b4b';
        c.beginPath();
        c.roundRect(x - botW * 0.6, y + height - 2, botW * 1.2, 6, [2, 2, 4, 4]);
        c.fill();

      } else {
        // Minimalist Pastel Terracotta Pot
        c.shadowColor = 'rgba(67, 85, 72, 0.1)';
        c.shadowBlur = 10;

        const pGrad = c.createLinearGradient(x - topW/2, y, x + topW/2, y + height);
        pGrad.addColorStop(0, '#e5a593');
        pGrad.addColorStop(0.6, '#d48d79');
        pGrad.addColorStop(1, '#be7560');

        // Trapezoid Body
        c.fillStyle = pGrad;
        c.beginPath();
        c.moveTo(x - topW * 0.46, y + rimH);
        c.lineTo(x + topW * 0.46, y + rimH);
        c.lineTo(x + botW * 0.5, y + height);
        c.lineTo(x - botW * 0.5, y + height);

        c.fill();

    function drawStemLeaf(c, x, y, len, angle, scale, flip = 1) {
      c.save();

      c.rotate(angle);
      c.scale(scale * flip, scale);
      c.beginPath();
      c.moveTo(0, 0);
      c.quadraticCurveTo(len * 0.45, -len * 0.35, len, 0);
      c.quadraticCurveTo(len * 0.45, len * 0.35, 0, 0);
      const g = c.createLinearGradient(0, -len * 0.2, len, 0);
      g.addColorStop(0, '#759472');
      g.addColorStop(1, '#97b192');
      c.fillStyle = g;
      c.fill();
      c.restore();
    }

    function renderGarden() {
      canvasEngine.sway += 0.025;
      canvasEngine.currentProgress += (canvasEngine.targetProgress - canvasEngine.currentProgress) * 0.06;
      const p = canvasEngine.currentProgress;

      const w = canvas.getBoundingClientRect().width;
      const h = canvas.getBoundingClientRect().height;
      ctx.clearRect(0, 0, w, h);

      const swayOffset = Math.sin(canvasEngine.sway) * (4 * Math.min(1, p * 1.5));
      const type = canvasEngine.flowerType;
      const potStyle = canvasEngine.potStyle;

      renderFlowerSpecies(ctx, type, potStyle, w, h, p, swayOffset);

      // Celestial Sparkle Particle Burst
      if (canvasEngine.sparkles.length > 0) {
        for (let i = canvasEngine.sparkles.length - 1; i >= 0; i--) {
          const s = canvasEngine.sparkles[i];
          s.x += s.vx;
          s.y += s.vy;
          s.alpha -= 0.02;
          ctx.fillStyle = s.color;
          ctx.globalAlpha = Math.max(0, s.alpha);
          ctx.beginPath();
          ctx.arc(s.x, s.y, s.size, 0, Math.PI * 2);
          ctx.fill();
          ctx.globalAlpha = 1.0;
          if (s.alpha <= 0) {
            canvasEngine.sparkles.splice(i, 1);
          }
        }
      }

      // Rain Droplets
      if (canvasEngine.droplets.length > 0) {
        ctx.strokeStyle = 'rgba(125, 185, 235, 0.75)';
        ctx.lineWidth = 2;
        const baseY = h * 0.84;
        for (let i = canvasEngine.droplets.length - 1; i >= 0; i--) {
          const d = canvasEngine.droplets[i];
          d.y += d.vy;
          ctx.beginPath();
          ctx.moveTo(d.x, d.y);
          ctx.lineTo(d.x, d.y + d.length);
          ctx.stroke();
          if (d.y > baseY) {
            canvasEngine.droplets.splice(i, 1);
          }
        }
      }

      requestAnimationFrame(renderGarden);
    }
